  וְיֵשׁ שֶׁיֵּשׁ לוֹ טְלָפַיִם וְאֵין לוֹ קַרְנַיִם

-
  גְּמָ'
-%%question%%%
  כֹּל שֶׁיֵּשׁ לוֹ קַשְׂקֶשֶׂת דָּג טָהוֹר
  יֵשׁ שֶׁיֵּשׁ לוֹ סְנַפִּיר וְאֵין לוֹ קַשְׂקֶשֶׂת דָּג טָמֵא
  מִכְּדֵי אֲנַן אַקַּשְׂקֶשֶׂת סָמְכִינַן סְנַפִּיר דִּכְתַב רַחֲמָנָא לְמָה לִי
-%%answer%%%
  אִי לָא כְּתַב רַחֲמָנָא סְנַפִּיר
  הֲוָה אָמֵינָא מַאי קַשְׂקֶשֶׂת דִּכְתִיב סְנַפִּיר וַאֲפִלּוּ דָּג טָמֵא
  כְּתַב רַחֲמָנָא סְנַפִּיר וְקַשְׂקֶשֶׂת
-%%question%%%
  וְהַשְׁתָּא דִּכְתַב רַחֲמָנָא סְנַפִּיר וְקַשְׂקֶשֶׂת מְנָלַן דְּקַשְׂקֶשֶׂת לְבוּשָׁא הוּא
-%%answer%%%
  דִּכְתִיב וְשִׁרְיוֹן קַשְׂקַשִּׂים הוּא לָבוּשׁ
-%%question%%%
  וְלִכְתּוֹב רַחֲמָנָא קַשְׂקֶשֶׂת וְלָא בָּעֵי סְנַפִּיר
-%%answer%%%
  אָמַר רַבִּי אֲבָהוּ וְכֵן תָּנָא דְּבֵי רַבִּי יִשְׁמָעֵאל יַגְדִּיל תּוֹרָה וְיַאְדִּיר
